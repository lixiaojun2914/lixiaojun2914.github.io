名字删掉之后，进程仍持有原来的网络环境
环境：7.0.0-44-generic，aarch64

$ sudo ip netns add nslab-a
$ sudo ip netns exec nslab-a sleep 300 >/dev/null 2>&1 &
[1] 66131
$ PID=$(sudo ip netns pids nslab-a | head -n 1)
$ echo "PID=$PID"
PID=66131

$ sudo readlink /proc/66131/ns/net
net:[4026532797]

$ sudo ip netns del nslab-a

$ sudo ip netns list
（无输出）

$ sudo readlink /proc/66131/ns/net
net:[4026532797]

$ sudo nsenter -t 66131 -n ip -br link
lo               DOWN           00:00:00:00:00:00 <LOOPBACK>

$ sudo kill 66131
